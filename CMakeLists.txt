cmake_minimum_required(VERSION 3.18)
project(warwarden_device LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
    set(Protobuf_USE_STATIC_LIBS ON)
endif()
find_package(Protobuf 3.15 REQUIRED)
find_package(Threads REQUIRED)
include(CTest)
protobuf_generate_cpp(DEVICE_PROTO_SRC DEVICE_PROTO_HDR external/warwarden-protocol/proto/mfs/icd/v1/mfs.proto)
add_library(device_contract STATIC ${DEVICE_PROTO_SRC} ${DEVICE_PROTO_HDR})
target_link_libraries(device_contract PUBLIC protobuf::libprotobuf)
target_include_directories(device_contract PUBLIC "${CMAKE_CURRENT_BINARY_DIR}")
# Hardware-independent shared runtime; the EFF target never links a LiDAR backend.
add_library(turret_runtime STATIC icd/runtime.cpp)
target_include_directories(turret_runtime PUBLIC icd external/LiDAR/capture PRIVATE motor)
target_link_libraries(turret_runtime PUBLIC device_contract Threads::Threads)
if(WIN32)
    target_link_libraries(turret_runtime PUBLIC ws2_32)
else()
    target_sources(turret_runtime PRIVATE motor/motor.cpp)
    target_compile_definitions(turret_runtime PRIVATE _DEFAULT_SOURCE)
endif()
add_executable(mfs_eff apps/eff.cpp)
target_link_libraries(mfs_eff PRIVATE turret_runtime)
option(BUILD_OBS "Build the OBS application with LiDAR capture" ON)
if(BUILD_OBS)
    add_subdirectory(external/LiDAR)
    add_executable(mfs_obs apps/obs.cpp)
    target_link_libraries(mfs_obs PRIVATE turret_runtime lidar_capture)
endif()
if(BUILD_TESTING)
    add_executable(laser_output_contract tests/laser_output.cpp)
    target_include_directories(laser_output_contract PRIVATE icd)
    target_link_libraries(laser_output_contract PRIVATE device_contract)
    if(WIN32)
        target_link_libraries(laser_output_contract PRIVATE ws2_32)
    endif()
    add_test(NAME laser_output_contract COMMAND laser_output_contract)
    add_executable(acquisition_contract tests/acquisition_contract.cpp)
    target_include_directories(acquisition_contract PRIVATE icd external/LiDAR/capture)
    add_test(NAME acquisition_contract COMMAND acquisition_contract)
endif()
if(BUILD_TESTING AND UNIX)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_library(motor_test_bridge SHARED tests/motor_test_bridge.cpp motor/motor.cpp)
    target_include_directories(motor_test_bridge PRIVATE motor)
    target_compile_definitions(motor_test_bridge PRIVATE _DEFAULT_SOURCE)
    add_test(NAME uart_motor COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_uart.py" $<TARGET_FILE:motor_test_bridge>)
endif()
